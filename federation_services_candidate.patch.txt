+# Native accounting bridge
+Candidate implementation. Requires native `account_payment`, with the provider addon
+and credentials configured separately. No webhook or independent ledger is added.
+
+Confirmed positive charges create one draft `account.move` under a locked event.
+The selected product owns income accounts and tax defaults. Amount is the line unit
+price, not a promised tax-inclusive total; finance must verify the resulting invoice
+before posting. Club/federation charge ownership is separate from invoice company.
+The payer is the invoice partner, never inferred from a guardian relationship.
+
+Native invoice portal checkout, payment transactions, reconciliation, credit notes
+and cancellation remain authoritative. Balance and payment state are related fields.
+An invoiced event cannot settle until its posted invoice is paid. Non-invoiced legacy
+events retain the existing handoff workflow. This does not certify provider sandbox
+callbacks, partial payments, refunds, multi-company rules or live reconciliation.

+from odoo import _, api, fields, models
+from odoo.exceptions import AccessError, ValidationError
+
+
+class Finance(models.Model):
+    _inherit = 'federation.finance.event'
+    account_move_id = fields.Many2one('account.move', readonly=True, copy=False, ondelete='restrict', check_company=True)
+    company_id = fields.Many2one('res.company', required=True, default=lambda self: self.env.company)
+    invoice_product_id = fields.Many2one('product.product', check_company=True)
+    charge_owner = fields.Selection([('federation', 'Federation'), ('club', 'Club')], default='federation', required=True)
+    invoice_currency_id = fields.Many2one(related='account_move_id.currency_id')
+    invoice_balance = fields.Monetary(related='account_move_id.amount_residual', currency_field='invoice_currency_id')
+    invoice_payment_state = fields.Selection(related='account_move_id.payment_state')
+
+    def _finance_actor(self):
+        if not self.env.user.has_group('account.group_account_invoice'):
+            raise AccessError(_('Invoice administration permission is required.'))
+        self.check_access('write')
+
+    def write(self, vals):
+        if vals.get('state') == 'settled' and any(e.account_move_id and (e.account_move_id.state != 'posted' or e.account_move_id.payment_state != 'paid') for e in self):
+            raise ValidationError(_('An unpaid invoice cannot be bypassed by a state write.'))
+        if vals.get('state') == 'cancelled' and any(e.account_move_id and e.account_move_id.state != 'cancel' for e in self):
+            raise ValidationError(_('Resolve the native invoice before cancelling its obligation.'))
+        if 'account_move_id' in vals:
+            raise AccessError(_('Create invoices through the finance command.'))
+        if {'company_id', 'currency_id', 'partner_id', 'amount', 'event_type', 'invoice_product_id', 'charge_owner'} & set(vals) and any(self.mapped('account_move_id')):
+            raise ValidationError(_('Use native invoice corrections after invoicing.'))
+        return super().write(vals)
+
+    @api.model_create_multi
+    def create(self, vals_list):
+        if any(v.get('account_move_id') for v in vals_list):
+            raise AccessError(_('Invoice links are command-owned.'))
+        return super().create(vals_list)
+
+    def action_create_invoice(self):
+        self.ensure_one()
+        self._finance_actor()
+        self.env.cr.execute('SELECT id FROM federation_finance_event WHERE id=%s FOR UPDATE', [self.id])
+        self.invalidate_recordset()
+        if self.account_move_id:
+            return self.action_open_invoice()
+        if self.state != 'confirmed' or self.event_type != 'charge' or self.amount <= 0:
+            raise ValidationError(_('Only confirmed positive charges can create customer invoices.'))
+        if not self.partner_id or not self.invoice_product_id:
+            raise ValidationError(_('Select the authorised payer and a configured accounting product.'))
+        if self.charge_owner == 'club' and not self.club_id:
+            raise ValidationError(_('A club charge needs a club.'))
+        if self.company_id not in self.env.companies:
+            raise AccessError(_('The invoice company is outside your scope.'))
+        move = self.env['account.move'].with_company(self.company_id).create({
+            'move_type': 'out_invoice', 'partner_id': self.partner_id.id,
+            'company_id': self.company_id.id, 'currency_id': self.currency_id.id,
+            'invoice_origin': self.name,
+            'invoice_line_ids': [fields.Command.create({'product_id': self.invoice_product_id.id,
+                'name': self.name, 'quantity': 1, 'price_unit': self.amount})]})
+        super().write({'account_move_id': move.id})
+        return self.action_open_invoice()
+
+    def action_open_invoice(self):
+        self.ensure_one()
+        self.account_move_id.check_access('read')
+        return {'type': 'ir.actions.act_window', 'res_model': 'account.move',
+                'res_id': self.account_move_id.id, 'view_mode': 'form', 'target': 'current'}
+
+    def action_settle(self):
+        for event in self:
+            if event.account_move_id and (event.account_move_id.state != 'posted' or event.account_move_id.payment_state != 'paid'):
+                raise ValidationError(_('Settle only after the native invoice is posted and fully paid.'))
+        return super().action_settle()
+
+    def action_cancel(self):
+        if any(e.account_move_id and e.account_move_id.state != 'cancel' for e in self):
+            raise ValidationError(_('Resolve the invoice or credit-note workflow in Accounting first.'))
+        return super().action_cancel()

+# Court Operations and Visual Planning
+
+Implementation candidate, not a staging acceptance certificate.
+
+## Installation
+Install or upgrade the supported `sports_federation_app` entry point. Its dependency graph now includes this addon. Apply the incremental patch only to the supplied current source baseline. Do not reapply earlier patches: their release-confidence, journey and duty-coverage code is already in that baseline.
+
+## Court operation
+1. Configure aggregate, period-total or sets-won scoring on the competition edition. Configure a winning set target where applicable.
+2. Publish through the existing independent schedule review and publication workflow.
+3. Open the canonical match-day session and assign named internal court operators.
+4. Operators open Match Day > Run this court, select their court, inspect lineups and separate accepted/eligible/present official facts, save drafts, then submit for result review.
+5. Submission is not approval. Existing verification, approval, corrections and standings controls remain authoritative.
+
+## Connectivity contract
+Offline capture is supported in an already loaded, authenticated workspace. This is not a cold-start offline PWA: opening the application or an uncached match requires connectivity. Device drafts and queued commands are stored under database/user/court/match-specific localStorage keys. No birth dates or contact details are cached. Use dedicated trusted devices; localStorage is not encrypted and a shared unlocked browser is not an access boundary.
+
+Each command carries an actor-bound content receipt, expected draft revision and schedule/result signature. Retry exactly the same key and payload after a lost response. A batch is atomic. A stale draft, changed publication, changed score, revoked operator or closed session stops synchronisation. No automatic score merging occurs. Export evidence before explicitly discarding the local queue or acknowledging a changed context. Exported evidence is not an authoritative result and is never automatically imported.
+
+Interrupted matches retain drafts for resumption. Disputed and abandoned captures raise linked incidents and retain a coordinator hold, even if an operator later changes the outcome back to normal. Match-day managers use Court recovery and exceptions > Resolve exception to record a reasoned resume or abandonment decision. Confirmed abandonment creates no official score. Forfeits require a reason and a score favouring the non-forfeiting team, then follow ordinary independent result review.
+
+## Visual planning
+Open a schedule and select Visual planning. The workspace projects existing stages, progression ranks, fixture sources and a court/time grid. Select a fixture and preview a manual move; applying it pins the assignment. Regeneration retains non-automatic assignments. Team/court unavailable windows and hard minimum rest are distinct from fairness preferences.
+
+Every preview is stored server-side, belongs to its actor and contains the exact candidate map. Before/after views show changed slots, unallocated fixtures, hard failures, warnings and weighted fairness penalty (lower is better). Apply revalidates the map and signature, performs one governed revision and retains the preview. Rejected-slot explanations do not misrepresent a heuristic solver as a proof of infeasibility.
+
+Published schedules cannot be edited here. Create a governed amendment, then return to the existing independent review/publication controls. Live operations must be closed before replacement publication. Practical constraints copy to amendment revisions.
+

+    def _access(self, matchday_id, court_id):
+        actor = self.env.user
+        if actor.share:
+            raise AccessError(_("Court operation requires a named internal account."))
+        access = self.env["federation.court.access"].sudo().search([
+            ("matchday_id", "=", int(matchday_id)), ("court_id", "=", int(court_id)),
+            ("user_id", "=", actor.id), ("active", "=", True)
+        ], limit=1)
+        if not access:
+            raise AccessError(_("You are not assigned to this court."))
+        session = access.matchday_id.active_session_id
+        if not session or session.state != "open":
+            raise ValidationError(_("Open the published match-day session before running this court."))
+        self.env.cr.execute("SELECT id FROM federation_court_access WHERE id=%s FOR SHARE", (access.id,))
+        access.invalidate_recordset()
+        if not access.active or access.user_id != actor:
+            raise AccessError(_("Court access was revoked."))
+        self.env.cr.execute("SELECT id FROM federation_matchday_session WHERE id=%s FOR SHARE", (session.id,))
+        session.invalidate_recordset()
+        if session.state != "open" or session.publication_id != access.matchday_id.current_publication_id:
+            raise ValidationError(_("The operational publication changed. Reopen the current session."))
+        return access, session

+    def _match(self, access, match_id):
+        match = self.env["federation.match"].sudo().browse(int(match_id)).exists()
+        if not match or match.schedule_publication_id.matchday_id != access.matchday_id:
+            raise AccessError(_("Match is outside this court."))
+        slot = match.operational_slot_id or match.published_slot_id
+        if not slot or slot.court_id != access.court_id or match.operational_status in {"cancelled", "postponed"}:
+            raise AccessError(_("Match is not currently runnable on this court."))
+        return match

+    @api.model
+    def sync(self, matchday_id, court_id, match_id, commands):
+        if not isinstance(commands, list) or not 1 <= len(commands) <= 100:
+            raise ValidationError(_("Synchronise 1 to 100 commands at a time."))
+        with self.env.cr.savepoint():
+            access, session = self._access(matchday_id, court_id)
+            match = self._match(access, match_id)
+            match._lock_result_transition()
+            match.invalidate_recordset()
+            capture = self._capture(match, self._signature(match, session))
+            capture.invalidate_recordset()
+            receipts = self.env["federation.court.receipt"].sudo().with_context(**{_CONTEXT: _TOKEN})
+            responses = []
+            for command in commands:
+                try:
+                    validate_command(command)
+                except ValueError as error:
+                    raise ValidationError(str(error)) from error
+                fingerprint = digest(command)
+                replay = receipts.search([("capture_id", "=", capture.id), ("key", "=", command["key"])], limit=1)
+                if replay:
+                    if replay.digest != fingerprint or replay.actor_id != self.env.user:
+                        raise ValidationError(_("Idempotency key was reused for different content or actor."))
+                    responses.append(replay.response)
+                    continue
+                if (command["revision"] != capture.revision or command["signature"] != self._signature(match, session)
+                        or capture.signature != command["signature"]):
+                    raise ValidationError(_("Conflict: schedule, result or draft changed. Keep local evidence and reconcile explicitly."))
+                if capture.submitted or capture.closed or match.result_state not in {"draft", "corrected"}:
+                    raise ValidationError(_("Submitted or official results cannot be overwritten from court capture."))
+                operation = command["operation"]
+                if operation == "undo":
+                    if not capture.undo_available:
+                        raise ValidationError(_("No saved draft is available to undo."))
+                    payload = capture.previous_payload
+                else:
+                    edition = match._competition_workflow_edition()
+                    try:
+                        payload = validate_capture(command["payload"], edition.court_scoring_mode,
+                                                   edition.court_set_target, edition.court_max_periods, complete=operation == "submit")
+                    except ValueError as error:
+                        raise ValidationError(str(error)) from error
+                values = {"payload": payload, "previous_payload": capture.payload,
+                          "undo_available": operation != "undo", "revision": capture.revision + 1,
+                          "signature": command["signature"]}
+                if payload["outcome"] in {"abandoned", "disputed"}:
+                    values["review_required"] = True
+                if operation == "submit":
+                    if capture.review_required:
+                        raise ValidationError(_("A coordinator must resolve the dispute or abandonment hold."))
+                    if payload["outcome"] in {"interrupted", "abandoned", "disputed"}:
+                        raise ValidationError(_("Save this exception for coordinator review; it is not an official result."))
+                    self._submit(match, payload)
+                    values.update({"submitted": True, "undo_available": False,
+                                   "signature": self._signature(match, session)})
+                capture.write(values)
+                if payload["outcome"] != "normal" and operation != "undo":
+                    self.env["federation.matchday.incident"].sudo().create({
+                        "matchday_id": access.matchday_id.id, "incident_type": "other", "court_capture_id": capture.id,
+                        "description": "%s | %s | %s | command %s" % (
+                            match.display_name, payload["outcome"], payload["reason"], command["key"])})
+                response = self._response(capture, match, session)
+                receipts.create({"capture_id": capture.id, "key": command["key"], "digest": fingerprint,
+                                 "actor_id": self.env.uid, "operation": operation, "response": response})
+                responses.append(response)
+            return {"receipts": responses, "current": self._response(capture, match, session)}
+
+    def _submit(self, match, payload):
+        # Authorization has already been checked against the named court operator.
+        target = match.with_user(self.env.user).sudo().with_context(**{
+            _RESULT_COMMAND_CONTEXT_KEY: _RESULT_COMMAND_TOKEN, _CONTEXT: _TOKEN})
+        target.write({"home_score": payload["home"], "away_score": payload["away"],
+                      "score_entry_confirmed": True, "zero_zero_confirmed": payload["zero_confirmed"],
+                      "court_outcome": payload["outcome"], "court_outcome_reason": payload["reason"]})
+        target.result_period_ids.unlink()
+        edition = target._competition_workflow_edition()
+        for index, period in enumerate(payload["periods"], 1):
+            home, away = period["home"], period["away"]
+            if edition.court_scoring_mode == "sets":
+                home, away = int(home > away), int(away > home)
+            target.env["federation.match.result.period"].create({"match_id": target.id, "sequence": index,
+                "home_score": home, "away_score": away,
+                "note": "Raw set: %s:%s" % (period["home"], period["away"])})
+        target.action_submit_result()
+
+    @api.model
+    def reconcile(self, matchday_id, court_id, match_id, revision, reason):
+        if not isinstance(reason, str) or not reason.strip() or len(reason) > 2000:
+            raise ValidationError(_("Record why the draft remains valid."))
+        with self.env.cr.savepoint():
+            access, session = self._access(matchday_id, court_id)
+            match = self._match(access, match_id)
+            match._lock_result_transition()
+            match.invalidate_recordset()
+            capture = self._capture(match, self._signature(match, session))
+            capture.invalidate_recordset()
+            if capture.revision != int(revision) or capture.submitted or match.result_state not in {"draft", "corrected"}:
+                raise ValidationError(_("The draft changed or is no longer mutable."))
+            capture.write({"signature": self._signature(match, session), "revision": capture.revision + 1})
+            response = self._response(capture, match, session)
+            self.env["federation.court.receipt"].sudo().with_context(**{_CONTEXT: _TOKEN}).create({
+                "capture_id": capture.id, "key": "reconcile:%s" % capture.revision,
+                "digest": digest({"reason": reason, "revision": capture.revision}), "actor_id": self.env.uid,
+                "operation": "reconcile", "response": {**response, "reason": reason.strip()}})
+            return response

+"""Pure, bounded input validation shared by online and offline commands."""
+import hashlib
+import json
+
+OUTCOMES = {"normal", "interrupted", "abandoned", "forfeit_home", "forfeit_away", "disputed"}
+MODES = {"aggregate", "periods", "sets"}
+
+
+def digest(value):
+    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":"),
+                                    ensure_ascii=True, allow_nan=False).encode()).hexdigest()
+
+
+def validate_capture(value, mode="aggregate", set_target=0, max_periods=100, complete=True):
+    if not isinstance(value, dict) or set(value) - {
+        "home", "away", "periods", "outcome", "reason", "zero_confirmed"
+    }:
+        raise ValueError("Unknown result fields.")
+    if mode not in MODES:
+        raise ValueError("Unknown scoring mode.")
+    result = {"home": value.get("home", 0), "away": value.get("away", 0),
+              "periods": value.get("periods", []), "outcome": value.get("outcome", "normal"),
+              "reason": value.get("reason", ""), "zero_confirmed": value.get("zero_confirmed", False)}
+    if result["outcome"] not in OUTCOMES:
+        raise ValueError("Unknown match outcome.")
+    if not isinstance(result["reason"], str) or len(result["reason"]) > 2000:
+        raise ValueError("Reason must be at most 2000 characters.")
+    result["reason"] = result["reason"].strip()
+    if result["outcome"] != "normal" and not result["reason"]:
+        raise ValueError("Record a reason for exceptional outcomes.")
+    if type(result["zero_confirmed"]) is not bool:
+        raise ValueError("Zero score confirmation must be a boolean.")
+    for key in ("home", "away"):
+        if type(result[key]) is not int or not 0 <= result[key] <= 100000:
+            raise ValueError("Scores must be non-negative bounded integers.")
+    periods = result["periods"]
+    if not isinstance(periods, list) or len(periods) > max_periods:
+        raise ValueError("Too many periods or invalid period data.")
+    for period in periods:
+        if not isinstance(period, dict) or set(period) != {"home", "away"}:
+            raise ValueError("Each period needs exactly two scores.")
+        if any(type(period[k]) is not int or not 0 <= period[k] <= 100000 for k in ("home", "away")):
+            raise ValueError("Invalid period score.")
+    if mode == "aggregate" and periods:
+        raise ValueError("This competition uses aggregate scoring.")
+    if complete and mode in {"periods", "sets"} and result["outcome"] == "normal" and not periods:
+        raise ValueError("Enter the sets or periods.")
+    if periods:
+        totals = (sum(p["home"] for p in periods), sum(p["away"] for p in periods))
+        if mode == "sets":
+            if complete and any(p["home"] == p["away"] for p in periods):
+                raise ValueError("A completed set cannot be tied.")
+            totals = (sum(p["home"] > p["away"] for p in periods),
+                      sum(p["away"] > p["home"] for p in periods))
+            if complete and set_target and result["outcome"] == "normal" and max(totals) != set_target:
+                raise ValueError("The winning set target has not been reached.")
+            if set_target:
+                wins = [0, 0]
+                for index, p in enumerate(periods):
+                    wins[0 if p["home"] > p["away"] else 1] += 1
+                    if max(wins) >= set_target and index != len(periods) - 1:
+                        raise ValueError("No sets may follow the winning set.")
+        if totals != (result["home"], result["away"]):
+            raise ValueError("Aggregate scores do not match periods or sets.")
+    if complete and result["outcome"] == "forfeit_home" and result["home"] >= result["away"]:
+        raise ValueError("Home forfeit requires an away winning result.")
+    if complete and result["outcome"] == "forfeit_away" and result["away"] >= result["home"]:
+        raise ValueError("Away forfeit requires a home winning result.")
+    return result
+
+
+def validate_command(command):
+    if not isinstance(command, dict) or set(command) != {"key", "revision", "operation", "payload", "signature"}:
+        raise ValueError("Malformed court command.")
+    if not isinstance(command["key"], str) or not 8 <= len(command["key"]) <= 80:
+        raise ValueError("A bounded idempotency key is required.")
+    if type(command["revision"]) is not int or command["revision"] < 0:
+        raise ValueError("Invalid revision.")
+    if command["operation"] not in {"save", "undo", "submit"}:
+        raise ValueError("Unknown court command.")
+    if not isinstance(command["signature"], str) or len(command["signature"]) != 64:
+        raise ValueError("A schedule and result signature is required.")
+    return command

+# Member and family services
+Implementation candidate, not an acceptance certificate.
+
+People are existing `res.partner` records; accounts remain `res.users`. The optional
+player `person_id` is manager-controlled and unique. Memberships are season/club/role
+records and do not replace licences, rosters or eligibility. No identity is inferred
+from an email address. Existing players require reviewed identity linking.
+
+Managers create and verify dated guardian or payer relationships. Only current
+verified guardian relationships expand personal document and membership access.
+Payer relationships never expand that access. Billing access remains native invoice
+partner access. `/my/family` exposes bounded dictionaries, not elevated recordsets.
+
+Requests use the shared upload policy and scanning configuration. Uploaded is not
+verified. Verification is independent and applies only to the named purpose; it
+never creates a parental approval or a licence. Existing parental approvals remain
+in their owning workflow. Renewal creates/reuses a draft membership, preserving its
+previous season reference; it does not activate membership or eligibility.
+
+Remaining work: reviewed identity backfill, duplicate review/merge workflow, linked
+parental-approval requests, notification routing, company-scoped administration,
+full browser and concurrent transaction tests. No claim of complete point 6.

+def compare_cohorts(previous, current):
+    previous, current = set(previous), set(current)
+    returning = previous & current
+    return {
+        'previous': len(previous), 'current': len(current),
+        'returning': len(returning), 'new': len(current - previous),
+        'not_returned': len(previous - current),
+        'retention_percent': round(100 * len(returning) / len(previous), 2) if previous else None,
+    }
